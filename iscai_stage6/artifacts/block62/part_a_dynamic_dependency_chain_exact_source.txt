def adb_shadow_geometry_from_xy(x_track, y_track, include_radial_zone=False):
    """
    Convert reconstructed x(t), y(t) into dynamic ADB shadow geometry.

    Coordinate convention:
        ego vehicle at (0, 0)
        +y = forward
        +x = right
        theta = atan2(x, y), in degrees

    Dynamic quantities:
        r(t)
        theta_c(t)
        theta_center(t) = theta_c(t) - Delta theta
        epsilon(t)
        theta_min(t)
        theta_max(t)
    """
    target_range = np.sqrt(x_track**2 + y_track**2)
    theta_c = np.rad2deg(np.arctan2(x_track, y_track))
    shadow_center = theta_c - delta_theta_deg
    effective_half_width = target_width_m / 2.0 + lateral_safety_margin_m
    epsilon = np.rad2deg(np.arctan(effective_half_width / np.maximum(target_range, EPS)))

    geometry = {
        "range": target_range,
        "range_target": target_range,
        "theta_c": theta_c,
        "shadow_center": shadow_center,
        "epsilon": epsilon,
        "theta_min": shadow_center - epsilon,
        "theta_max": shadow_center + epsilon,
    }

    if include_radial_zone:
        geometry["r_min"] = target_range + adb_trz_front_margin_units
        geometry["r_max"] = geometry["r_min"] + adb_trz_transition_length_units

    return geometry

def adb_intensity_map(theta_min_deg, theta_max_deg, x_m, y_m):
    # Current distance from ego vehicle to target center
    target_range_m = np.hypot(x_m, y_m)
    # Same logic as the simulation 1's example intensity map
    r_min_m = target_range_m + radial_safety_margin_m
    r_max_m = r_min_m + transition_length_m

    intensity = np.ones_like(adb_range_grid_m, dtype=float)

    inside_shadow = (
        (adb_angle_grid_deg >= theta_min_deg)
        & (adb_angle_grid_deg <= theta_max_deg)
    )

    full_shadow = inside_shadow & (adb_range_grid_m <= r_min_m)
    transition = (
        inside_shadow
        & (adb_range_grid_m > r_min_m)
        & (adb_range_grid_m < r_max_m)
    )
    intensity[full_shadow] = 0.0

    u = (adb_range_grid_m[transition] - r_min_m) / (r_max_m - r_min_m)
    intensity[transition] = 0.5 * (1.0 - np.cos(np.pi * u))

    return intensity

def adb_combine_intensity_maps(theta_min_values, theta_max_values, x_values, y_values):
    """Combine ADB intensity maps from multiple vehicles."""

    combined = np.ones_like(adb_range_grid_m, dtype=float)

    for theta_min_deg, theta_max_deg, x_m, y_m in zip(
        theta_min_values,
        theta_max_values,
        x_values,
        y_values
    ):
        vehicle_map = adb_intensity_map(theta_min_deg, theta_max_deg, x_m, y_m)
        combined = np.minimum(combined, vehicle_map)

    return combined
